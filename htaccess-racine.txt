# ── 1. FORCER HTTPS ─────────────────────────────────────
RewriteEngine On
RewriteCond %{HTTPS} off
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# ── 2. INTERDIRE LE DIRECTORY LISTING ───────────────────
Options -Indexes

# ── 3. BLOQUER L'ACCÈS AUX FICHIERS SENSIBLES ──────────
# Fichiers de gouvernance / documentation
<FilesMatch "\.(md|sql|txt|log|bak|old|orig|save|swp|tmp)$">
  Require all denied
</FilesMatch>

# Fichiers cachés (.git, .env, .htpasswd, etc.)
<FilesMatch "^\.">
  Require all denied
</FilesMatch>

# Fichiers de config spécifiques
<FilesMatch "^(SESSION_STATE|NOYAU|JOURNAL|CHANTIER|CTX_|PLAN_|GUIDE_|PERSONAS|BIBLE_|SUPABASE_|TEMPLATE_|MODULE_DEPENDENCY)">
  Require all denied
</FilesMatch>

# admin-test.html — page diagnostic
<Files "admin-test.html">
  Require all denied
</Files>

# admin-memo.html — mémo admin
<Files "admin-memo.html">
  Require all denied
</Files>

# ── 4. HEADERS DE SÉCURITÉ ─────────────────────────────
<IfModule mod_headers.c>
  # Empêcher le clickjacking
  Header always set X-Frame-Options "SAMEORIGIN"

  # Empêcher le MIME-type sniffing
  Header always set X-Content-Type-Options "nosniff"

  # Protection XSS navigateur
  Header always set X-XSS-Protection "1; mode=block"

  # Referrer : envoyer l'origine seulement vers même origine
  Header always set Referrer-Policy "strict-origin-when-cross-origin"

  # Permissions Policy — désactiver les APIs inutiles
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=()"

  # Content Security Policy
  # Autorise : self + CDN Bootstrap/BI/Supabase/menywise + Quill + inline scripts
  Header always set Content-Security-Policy "\
    default-src 'self'; \
    script-src 'self' 'unsafe-inline' https://cdn.jsdelivr.net https://cdnjs.cloudflare.com https://unpkg.com; \
    style-src 'self' 'unsafe-inline' https://cdn.jsdelivr.net https://cdnjs.cloudflare.com https://fonts.googleapis.com; \
    font-src 'self' https://cdn.jsdelivr.net https://fonts.gstatic.com; \
    img-src 'self' data: https://*.supabase.co; \
    connect-src 'self' https://*.supabase.co wss://*.supabase.co; \
    frame-ancestors 'self'; \
    base-uri 'self'; \
    form-action 'self'; \
    object-src 'none';\
  "

  # Strict-Transport-Security (HSTS) — 1 an
  Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"

  # Empêcher la mise en cache des pages HTML contenant des données sensibles
  <FilesMatch "\.html$">
    Header set Cache-Control "no-cache, no-store, must-revalidate"
    Header set Pragma "no-cache"
    Header set Expires "0"
  </FilesMatch>

  # Cache long sur les assets statiques (CSS, JS, images)
  <FilesMatch "\.(css|js|png|jpg|jpeg|gif|svg|ico|woff2?)$">
    Header set Cache-Control "public, max-age=2592000"
  </FilesMatch>
</IfModule>

# ── 5. PROTECTION HOTLINKING ───────────────────────────
<IfModule mod_rewrite.c>
  RewriteCond %{HTTP_REFERER} !^$ [NC]
  RewriteCond %{HTTP_REFERER} !^https?://(www\.)?manuelrohaut\.fr [NC]
  RewriteCond %{HTTP_REFERER} !^https?://localhost [NC]
  RewriteRule \.(png|jpg|jpeg|gif|svg|webp)$ - [F,NC,L]
</IfModule>

# ── 6. BLOQUER LES DOSSIERS INTERNES ───────────────────
# 00_GOUVERNANCE — bloqué par son propre .htaccess + ici en backup
<IfModule mod_rewrite.c>
  RewriteRule ^00_GOUVERNANCE(/|$) - [F,L]
  RewriteRule ^01_TEMPLATES(/|$) - [F,L]
  RewriteRule ^02_INFRASTRUCTURE(/|$) - [F,L]
  RewriteRule ^OLD_GOUVERNANCE(/|$) - [F,L]
  RewriteRule ^DATA_METIER(/|$) - [F,L]
</IfModule>

# ── 7. PAGES D'ERREUR PERSONNALISÉES ───────────────────
ErrorDocument 403 /bdb/403.html
ErrorDocument 404 /bdb/404.html

# ── 8. COMPRESSION GZIP ────────────────────────────────
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css text/javascript
  AddOutputFilterByType DEFLATE application/javascript application/json
  AddOutputFilterByType DEFLATE image/svg+xml
</IfModule>
